import streamlit as st import pandas as pd import requests import json from datetime import datetime import plotly.express as px # 페이지 기본 설정 st.set_page_config( page_title="팀 예산 관리 시스템", page_icon="📊", layout="wide", initial_sidebar_state="expanded" ) st.sidebar.title("⚙️ 시스템 설정") st.sidebar.markdown("스프레드시트 DB와 연결하기 위한 웹 앱 URL을 입력하세요.") gas_url = st.sidebar.text_input("Apps Script Web App URL", type="password", help="구글 앱스스크립트에서 배포 후 생성된 URL을 입력하세요.") with st.sidebar.expander("📖 구글 시트 연동 방법 가이드"): st.markdown(""" **1. 구글 스프레드시트 준비** - 새 구글 시트를 만들고 시트 이름을 `Sheet1`로 지정합니다. - 첫 번째 행(A1:F1)에 다음 헤더를 입력합니다: `id`, `member`, `date`, `category`, `amount`, `timestamp` **2. 앱스 스크립트(GAS) 설정** - 시트 메뉴에서 `확장 프로그램` > `Apps Script` 클릭 - 아래 코드를 복사해서 `코드.gs`에 붙여넣습니다. \`\`\`javascript const SHEET_NAME = 'Sheet1'; function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const data = JSON.parse(e.postData.contents); sheet.appendRow([ data.id, data.member, data.date, data.category, data.amount, new Date().toISOString() ]); return ContentService.createTextOutput(JSON.stringify({status: 'success'})) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({status: 'error'})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); const headers = rows[0]; const data = rows.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } ``` **3. API 배포하기** - 스크립트 편집기 우측 상단 `배포` > `새 배포` - 유형: `웹 앱` (톱니바퀴 아이콘) - 액세스 권한: **`모든 사용자`** - 배포 후 생성된 **웹 앱 URL**을 위 입력창에 붙여넣으세요. """) @st.cache_data(ttl=5) # 5초 캐시 유지 (빈번한 API 호출 방지) def load_data(url): """Apps Script를 통해 구글 시트 데이터를 가져옵니다.""" if not url: return pd.DataFrame() try: response = requests.get(url, timeout=10) if response.status_code == 200: data = response.json() if data: df = pd.DataFrame(data) df['amount'] = pd.to_numeric(df['amount']) # 최신 데이터가 위로 오도록 역순 정렬 return df.iloc[::-1].reset_index(drop=True) # 데이터가 비어있을 경우 빈 구조 반환 return pd.DataFrame(columns=['id', 'member', 'date', 'category', 'amount', 'timestamp']) except Exception as e: st.error("데이터를 불러오는 중 오류가 발생했습니다. URL을 확인해주세요.") return pd.DataFrame() def save_data(url, data): """Apps Script로 새 데이터를 전송하여 구글 시트에 기록합니다.""" try: headers = {'Content-Type': 'application/json'} response = requests.post(url, data=json.dumps(data), headers=headers, timeout=10) if response.status_code == 200 and response.json().get('status') == 'success': return True return False except Exception as e: st.error("데이터 저장 중 통신 오류가 발생했습니다.") return False st.title("📊 팀 예산 관리 시스템") st.markdown("Streamlit과 Google Sheets를 활용한 부장님 보고용 예산 취합 대시보드") if not gas_url: st.info("👈 왼쪽 사이드바에 **Apps Script Web App URL**을 입력하여 데이터베이스를 연결해주세요.") st.stop() # URL이 없으면 렌더링 중단 # 데이터 로드 with st.spinner('구글 시트에서 데이터를 불러오는 중...'): df = load_data(gas_url) # 탭 구성 tab1, tab2 = st.tabs(["📝 데이터 입력", "📈 전체 대시보드"]) with tab1: col1, col2 = st.columns([1, 2]) # 폼 영역 with col1: st.subheader("내역 입력") default_members = ["부장님", "팀원1", "팀원2", "팀원3", "팀원4"] if not df.empty and 'member' in df.columns: unique_members = [m for m in df['member'].unique() if m] for m in unique_members: if m not in default_members: default_members.append(m) default_members.append("➕ 새 팀원 추가 (직접 입력)") with st.form("budget_form", clear_on_submit=True): selected_member = st.selectbox("팀원 선택", default_members) new_member = st.text_input("새 팀원 이름 (위에서 '새 팀원 추가' 선택 시 입력)") date_val = st.date_input("해당 일자") category = st.selectbox("예산 항목", ["수선유지비", "비품", "개량공사"]) amount = st.number_input("사용 금액 (원)", min_value=0, step=1000, format="%d") submitted = st.form_submit_button("기록 저장하기", use_container_width=True) if submitted: final_member = new_member if selected_member == "➕ 새 팀원 추가 (직접 입력)" else selected_member if amount <= 0: st.error("금액은 0보다 큰 숫자를 입력해야 합니다.") elif not final_member.strip(): st.error("팀원 이름을 입력해주세요.") elif not gas_url: st.error("설정에서 Web App URL을 먼저 입력하세요.") else: new_entry = { "id": int(datetime.now().timestamp() * 1000), # 고유 ID "member": final_member.strip(), "date": date_val.strftime("%Y-%m-%d"), "category": category, "amount": amount } if save_data(gas_url, new_entry): st.success("데이터가 성공적으로 구글 시트에 기록되었습니다!") load_data.clear() # 캐시 초기화 st.rerun() # 화면 새로고침 # 최근 내역 영역 with col2: st.subheader("📂 최근 입력 내역 (실시간)") if df.empty: st.info("아직 기록된 예산 데이터가 없습니다.") else: # 화면 출력용 데이터 프레임 생성 display_df = df[['date', 'member', 'category', 'amount', 'timestamp']].copy() # 타임스탬프 포맷팅 등 깔끔하게 정리 display_df.columns = ['일자', '팀원', '항목', '사용 금액', '입력 시간'] # Streamlit Dataframe 렌더링 (금액 컬럼 포맷 지원) st.dataframe( display_df, column_config={ "사용 금액": st.column_config.NumberColumn("사용 금액", format="%d 원") }, use_container_width=True, hide_index=True, height=400 ) with tab2: if df.empty: st.warning("데이터가 없어 대시보드를 표시할 수 없습니다. 데이터를 먼저 입력해주세요.") else: # 요약 지표 (Metrics) st.subheader("핵심 지표 요약") m_col1, m_col2, m_col3 = st.columns(3) total_amount = df['amount'].sum() m_col1.metric("전체 누적 사용액", f"{total_amount:,.0f} 원") # 카테고리별 합계 중 가장 큰 항목 찾기 cat_sum = df.groupby('category')['amount'].sum() top_cat = cat_sum.idxmax() top_cat_amount = cat_sum.max() m_col2.metric("가장 많이 지출한 항목", top_cat, f"{top_cat_amount:,.0f} 원 지출") m_col3.metric("총 데이터 등록 건수", f"{len(df)} 건") st.divider() # 시각화 (Charts) c_col1, c_col2 = st.columns(2) with c_col1: st.markdown("**🏠 항목별 예산 분포**") fig_pie = px.pie( df, values='amount', names='category', hole=0.4, color_discrete_sequence=px.colors.qualitative.Pastel ) fig_pie.update_layout(margin=dict(t=20, b=20, l=0, r=0)) st.plotly_chart(fig_pie, use_container_width=True) with c_col2: st.markdown("**👥 팀원별 누적 사용액**") member_agg = df.groupby('member')['amount'].sum().reset_index() fig_bar = px.bar( member_agg, x='member', y='amount', text_auto='.2s', color='member', color_discrete_sequence=px.colors.qualitative.Set2 ) fig_bar.update_layout(margin=dict(t=20, b=20, l=0, r=0), showlegend=False) st.plotly_chart(fig_bar, use_container_width=True) st.divider() # 피벗 테이블 (월별/항목별 크로스탭) st.markdown("**📅 월별/항목별 요약 테이블 (취합본)**") df['month'] = df['date'].astype(str).str[:7] # 피벗테이블 생성 pivot_df = pd.pivot_table( df, values='amount', index='month', columns='category', aggfunc='sum', fill_value=0 ) # 합계 컬럼 추가 및 월 역순 정렬 pivot_df['총 합계'] = pivot_df.sum(axis=1) pivot_df = pivot_df.sort_index(ascending=False) # Streamlit 화면에 표시 (숫자 포맷팅) styled_pivot = pivot_df.style.format("{:,.0f}").background_gradient(cmap='Blues', subset=['총 합계']) st.dataframe(styled_pivot, use_container_width=True)
